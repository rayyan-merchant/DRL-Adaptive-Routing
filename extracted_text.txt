SECTION 1 — PROJECT OVERVIEW & DESIGN DECISIONS

1. Project Overview & Design Decisions
This project implements a Deep Reinforcement Learning (DRL) adaptive routing system in ns-3. A single
centralized DQN agent — directly inspired by DRSIR (Casas-Velasco et al., IEEE TNSM 2021) — learns
to select optimal paths through a 6-node simulated network, outperforming static Dijkstra routing under
congestion and link failures. The agent is then upgraded to DDQN for improved Q-value estimation.

SECTION 2 — TOPOLOGY DESIGN

2. Topology Design
The 6-node topology is intentionally heterogeneous. If all links had identical capacity and delay, Dijkstra
and DQN would always choose the same path and comparison would be meaningless. The design creates
scenarios where the shortest path by hop-count is NOT optimal under congestion — which is exactly what
the DQN agent must learn to exploit.

propagation delay.

import xml.etree.ElementTree as ET
import pandas as pd
def parse_flowmon(xml_file, duration=99.0):
tree = ET.parse(xml_file)
root = tree.getroot()
rows = []
for flow in root.iter("Flow"):
tx = int(flow.get("txPackets", 0))
rx = int(flow.get("rxPackets", 0))
loss_ratio = (tx-rx)/tx if tx>0 else 0
# delaySum is in nanoseconds (e.g. "12345678ns")
ds = float(flow.get("delaySum","0ns").replace("ns",""))
avg_delay_ms = (ds/rx/1e6) if rx>0 else 0
rxb = int(flow.get("rxBytes", 0))
tput_mbps = (rxb*8)/(duration*1e6)
rows.append({"flow": flow.get("flowId"),
"throughput_mbps": tput_mbps,
"avg_delay_ms": avg_delay_ms,
"loss_ratio": loss_ratio})
return pd.DataFrame(rows)

6.2 Observation Space Design
The observation vector has 10 elements. The neural network uses ONLY element 0 (SD pair index).

exactly.

6.5 Reward Implementation (env/metrics.py)
import numpy as np
def compute_reward(obs, action, k=3, beta=(1.0,1.0,1.0)):
"""
Compute DRSIR cost for chosen action. Agent MINIMIZES this.
obs: full 10-element observation vector
action: integer in [0, k-1]
"""
# Extract path metrics: shape (k, 3) — columns: [bw, delay, loss]
path_metrics = obs[1:].reshape(k, 3).astype(np.float32)
# Min-max normalize each column across k paths
norm = np.zeros_like(path_metrics)
for col in range(3):
vals = path_metrics[:, col]
vmin, vmax = vals.min(), vals.max()
if (vmax - vmin) < 1e-8:
norm[:, col] = 0.5 # all paths equal for this metric

else:
norm[:, col] = (vals - vmin) / (vmax - vmin)
bw_n, dl_n, ls_n = norm[action]
# DRSIR Equation 5: inversely proportional to BW, directly to delay+loss
reward = beta[0]*(1.0/(bw_n+1e-6)) + beta[1]*dl_n + beta[2]*ls_n
return float(reward)

import torch, torch.nn as nn, torch.nn.functional as F
from configs.hyperparams import HIDDEN_NEURONS, K_PATHS, N_SD_PAIRS
class QNetwork(nn.Module):
"""Online and Target network — identical architecture."""
def __init__(self).
super().__init__()
# Embed SD pair index into hidden space
self.embed = nn.Embedding(N_SD_PAIRS, HIDDEN_NEURONS)
self.fc1 = nn.Linear(HIDDEN_NEURONS, HIDDEN_NEURONS)
self.out = nn.Linear(HIDDEN_NEURONS, K_PATHS)
# Glorot (Xavier) uniform initialisation — from DRSIR
nn.init.xavier_uniform_(self.fc1.weight)
nn.init.xavier_uniform_(self.out.weight)
def forward(self, idx):
"""idx: LongTensor shape (batch,) → Q-values shape (batch, k)"""
x = self.embed(idx) # (batch, hidden)
x = F.relu(self.fc1(x)) # (batch, hidden)
return self.out(x) # (batch, k)
def best_action(self, idx, device):
"""Greedy action — argmin because we MINIMISE cost."""
with torch.no_grad():
t = torch.tensor([idx], dtype=torch.long).to(device)
return self.forward(t).argmin().item()

from collections import deque
import random, numpy as np
class ReplayBuffer:
def __init__(self, capacity):
self.buf = deque(maxlen=capacity)
def push(self, s, a, r, ns, done):
self.buf.append((s, a, r, ns, done))
def sample(self, n):
batch = random.sample(self.buf, n)
s,a,r,ns,d = zip(*batch)
return (np.array(s), np.array(a),
np.array(r,dtype=np.float32),
np.array(ns), np.array(d,dtype=np.float32))
def __len__(self): return len(self.buf)

class DQNAgent:
def __init__(self, device="cpu"):
self.device = device
self.online = QNetwork().to(device)
self.target = QNetwork().to(device)
self.target.load_state_dict(self.online.state_dict())
self.target.eval()
self.optim = optim.Adam(self.online.parameters(), lr=LR)
self.buffer = ReplayBuffer(BUFFER_SIZE)
self.steps = 0
self.epsilon = EPS_MAX
self.losses = []
def act(self, state_idx):
"""Epsilon-greedy action selection."""
self.epsilon = max(EPS_MIN, EPS_MAX - self.steps * DECAY_RATE)
if random.random() < self.epsilon:
return random.randint(0, K_PATHS-1) # explore
return self.online.best_action(state_idx, self.device) # exploit
def store(self, s, a, r, ns, done):
self.buffer.push(s, a, r, ns, done)
self.steps += 1
def train_step(self):
if len(self.buffer) < REPLAY_START: return None
s,a,r,ns,d = self.buffer.sample(BATCH_SIZE)
S = torch.LongTensor(s).to(self.device)
A = torch.LongTensor(a).to(self.device)
R = torch.FloatTensor(r).to(self.device)
NS = torch.LongTensor(ns).to(self.device)
D = torch.FloatTensor(d).to(self.device)
# Current Q-values from Online Net
q_pred = self.online(S).gather(1, A.unsqueeze(1)).squeeze(1)
# DQN target: R + gamma * min_a Q_target(s_next, a)
with torch.no_grad():
q_next = self.target(NS).min(1)[0] # min because we MINIMISE
q_target = R + GAMMA * q_next * (1 - D)
loss = nn.MSELoss()(q_pred, q_target)
self.optim.zero_grad(); loss.backward(); self.optim.step()
self.losses.append(loss.item())
if self.steps % TARGET_UPDATE == 0:
self.target.load_state_dict(self.online.state_dict())
return loss.item()
def save(self, path):
torch.save(self.online.state_dict(), path)
def load(self, path):
self.online.load_state_dict(torch.load(path))
self.target.load_state_dict(self.online.state_dict())

DDQN Upgrade — One Formula Change, Full Comparison

DDQN fixes DQN's overestimation bias. Since we minimize cost, overestimation means the agent thinks a
bad path is better than it is. The fix separates action selection (Online NN) from action evaluation (Target
NN). Everything else stays identical — same architecture, same buffer, same hyperparams.

9.1 DQN vs DDQN Target Formula

Target Q

DQN (current)

DDQN (upgrade)

R + γ · min_a Q_target(s', a)

R + γ · Q_target(s', argmin_a Q_online(s', a))

Who selects action Target NN (same net selects AND evaluates)

Online NN selects; Target NN evaluates only

Python code change q_next.min(1)[0]

q_next=target(NS).gather(1,best).squeeze(1)

Everything else

Identical network, buffer, epsilon, hyperparams

class DDQNAgent(DQNAgent):
"""DDQN: overrides only train_step(). Everything else inherited."""
def train_step(self):
if len(self.buffer) < REPLAY_START: return None
s,a,r,ns,d = self.buffer.sample(BATCH_SIZE)
S = torch.LongTensor(s).to(self.device)
A = torch.LongTensor(a).to(self.device)

NS = torch.LongTensor(ns).to(self.device)
D = torch.FloatTensor(d).to(self.device)
q_pred = self.online(S).gather(1, A.unsqueeze(1)).squeeze(1)
with torch.no_grad():
# DDQN: Online NN selects the best action
best = self.online(NS).argmin(1, keepdim=True)
# DDQN: Target NN evaluates ONLY that selected action
q_next = self.target(NS).gather(1, best).squeeze(1)
q_target = R + GAMMA * q_next * (1 - D)
loss = nn.MSELoss()(q_pred, q_target)
self.optim.zero_grad(); loss.backward(); self.optim.step()
self.losses.append(loss.item())
if self.steps % TARGET_UPDATE == 0:
